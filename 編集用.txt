const store = {
  mem: {},
  async load(key){
    if (typeof window.storage !== "undefined" && window.storage?.get){
      try{ const r = await window.storage.get(key); return r ? JSON.parse(r.value) : null; }catch(e){ return null; }
    }
    try{ const v = localStorage.getItem(key); return v ? JSON.parse(v) : null; }catch(e){ return this.mem[key] || null; }
  },
  async save(key, data){
    this.mem[key] = data;
    if (typeof window.storage !== "undefined" && window.storage?.set){
      try{ await window.storage.set(key, JSON.stringify(data)); return; }catch(e){}
    }
    try{ localStorage.setItem(key, JSON.stringify(data)); }catch(e){}
  }
};

async function apiGet(url){
  const r = await fetch(url || gasUrl);
  if(!r.ok) throw new Error("http " + r.status);

}
async function apiPost(action, payload){
  const r = await fetch(gasUrl, { method:"POST", body: JSON.stringify({ action, payload }) });
  if(!r.ok) throw new Error("http " + r.status);
  return await r.json();
}
function applyRemote(d){

}

  }

function fileToIcon(file){
  return new Promise((res, rej)=>{
    const img = new Image();
    const url = URL.createObjectURL(file);
    img.onload = ()=>{
      const size = 128;
      const c = document.createElement("canvas");
      c.width = c.height = size;
      const ctx = c.getContext("2d");
      const s = Math.min(img.width, img.height);
      ctx.drawImage(img, (img.width-s)/2, (img.height-s)/2, s, s, 0, 0, size, size);
      URL.revokeObjectURL(url);
      let q = .82, data = c.toDataURL("image/jpeg", q);
      while(data.length > 45000 && q > .3){ q -= .15; data = c.toDataURL("image/jpeg", q); }
      res(data);
    };
    img.onerror = ()=>{ URL.revokeObjectURL(url); rej(new Error("read error")); };
    img.src = url;
  });
}

function totals(){
  const map = {};
  S.members.forEach(m => map[m.id] = { ...m, gold:0, silver:0 });
  S.history.forEach(h => {
    if(!map[h.mid]) return;
    if(h.kind === "gold") map[h.mid].gold += h.pt; else map[h.mid].silver += h.pt;
  });
  return Object.values(map)
    .map(m => ({...m, total:m.gold+m.silver}))
    .sort((a,b)=> b.total - a.total || a.name.localeCompare(b.name,"ja"));
}

      if(inp.dataset.f==="label") r.label = inp.value; else r.min = Math.max(0,+inp.value||0);
      await sync("saveSettings", { settings: S.settings });
      renderRanking();
    };
  });
  document.getElementById("perksInput").value = S.settings.perks || "";
  document.getElementById("gasUrl").value = gasUrl || "";
}

/* ---- 閲覧専用モード（共有リンク） ---- */
async function tryViewerMode(){
  /* シート連携リンク: #view=GASのURL → 常に最新を表示 */
  const mv = location.hash.match(/#view=(.+)/);
  if(mv){
    document.getElementById("tabs").style.display = "none";
    document.getElementById("shareBtnBox").style.display = "none";
    try{
      const d = await apiGet(decodeURIComponent(mv[1]));
      applyRemote(d);
      renderRanking(totals(), `${new Date().toLocaleString("ja-JP")} 時点`);
    }catch(e){
      document.getElementById("top3").innerHTML = '<div class="empty">データを読み込めませんでした。通信環境を確認して再読み込みしてください。</div>';
    }
    return true;
  }
  /* データ埋め込みリンク（シート未連携時の従来方式） */
  const md = location.hash.match(/#d=(.+)/);
  if(md){
    try{
      const json = decodeURIComponent(escape(atob(decodeURIComponent(md[1]))));
      const p = JSON.parse(json);
      S.settings.perks = p.perks || "";
      S.settings.ranks = p.ranks || S.settings.ranks;
      document.getElementById("tabs").style.display = "none";
      document.getElementById("shareBtnBox").style.display = "none";
      renderRanking(p.list, p.updated ? `${p.updated} 更新` : "");
      return true;
    }catch(e){ return false; }
  }
  return false;
}

      reason: document.getElementById("gReason").value.trim(),

    document.getElementById("cAmount").value = "";

    renderAll();

      gasUrl = "";
      await store.save(CONF_KEY, { gasUrl:"" });
      setSyncBadge(false);

    }

      gasUrl = url;
      await store.save(CONF_KEY, { gasUrl });
      /* シートが空でローカルにデータがあれば移行を提案 */
      if((!d.members || !d.members.length) && S.members.length){
        if(confirm("シートはまだ空です。今このツールに入っているデータをシートに書き込みますか？")){
          const pushed = await apiPost("bulk", {

          applyRemote(d);
        }

      }
      await store.save(KEY, S);
      setSyncBadge(true);
      st.className = "gas-status ok"; st.textContent = "✓ 連携完了。シートにデータが記録されます";
      renderAll();

    }
  };

        const d = JSON.parse(r.result);

      }catch(err){ toast("読み込みに失敗しました（ファイル形式を確認）"); }
    };
    r.readAsText(f);
  };
}

/* ---- 起動 ---- */
(async function init(){
  document.getElementById("gDate").value = todayStr();
  if(await tryViewerMode()) return;

  const conf = await store.load(CONF_KEY);
  gasUrl = conf?.gasUrl || "";

  if(gasUrl){
    try{
      const d = await apiGet();
      applyRemote(d);
      await store.save(KEY, S);
      setSyncBadge(true);
    }catch(e){
      const saved = await store.load(KEY);
      if(saved && saved.members && saved.settings) S = saved;
      setSyncBadge(false);
      toast("⚠ シートに接続できないためローカルデータを表示中");
    }
  } else {
    const saved = await store.load(KEY);
    if(saved && saved.members && saved.settings) S = saved;
    setSyncBadge(false);
  }
  bind();
  renderAll();
})();
